// Generic feature-enable helper used by counter access control.

enum FeatureEnabledResult = {
FEATURE_ENABLED,
FEATURE_ILLEGAL,
FEATURE_VIRTUAL
}

// Several features are controlled by enable bits in machine,
// supervisor, and hypervisor CSRs.

function feature_enabled_for_priv(
p : Privilege,
machine_enable_bit : bits(1),
supervisor_enable_bit : bits(1),
hypervisor_enable_bit : bits(1)
) -> FeatureEnabledResult =
match p {
Machine =>
FEATURE_ENABLED,

```
Supervisor =>
  if machine_enable_bit == 0b1
  then FEATURE_ENABLED
  else FEATURE_ILLEGAL,

User =>
  if machine_enable_bit == 0b1
     & (not(currentlyEnabled(Ext_S)) | supervisor_enable_bit == 0b1)
  then FEATURE_ENABLED
  else FEATURE_ILLEGAL,

VirtualSupervisor =>
  if machine_enable_bit == 0b0
  then FEATURE_ILLEGAL
  else if hypervisor_enable_bit == 0b0
  then FEATURE_VIRTUAL
  else FEATURE_ENABLED,

VirtualUser =>
  if machine_enable_bit == 0b0
  then FEATURE_ILLEGAL
  else if hypervisor_enable_bit == 0b0 | supervisor_enable_bit == 0b0
  then FEATURE_VIRTUAL
  else FEATURE_ENABLED,
```

}

// Backward-compatible wrapper around feature_enabled_for_priv for callers
// that do not yet distinguish illegal-instruction and virtual-instruction
// exceptions.

function feature_enabled_for_priv_bool(
p : Privilege,
m : bits(1),
s : bits(1),
h : bits(1)
) -> bool =
feature_enabled_for_priv(p, m, s, h) == FEATURE_ENABLED
