# Test Case Rule: RULE 1: When TVM=1, attempting to execute an SFENCE.VMA instruction while executing in S-mode raises an illegal-instruction exception.

li t0, 1 << 20
csrs mstatus, t0
call machine_to_supervisor
sfence.vma x0, x0
# Result: mcause = 2

# Test Case Rule: RULE 2: When TVM=0, attempting to execute an SFENCE.VMA instruction is permitted in S-mode.

csrc mstatus, t0

sfence.vma x0, x0
li a0, 1
# Result: a0 = 1
